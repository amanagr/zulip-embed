import {Tabs, TabItem, Code} from "@astrojs/starlight/components";

The fastest way to embed Zulip chat. Copy-paste the snippet below, point
`server` at your instance, and paste a short-lived JWT into `auth-token`.

<Tabs>
  <TabItem label="HTML / CDN">
    ```html
    <script
      type="module"
      src="https://cdn.jsdelivr.net/npm/zulip-embed@1/dist/zulip-embed.js"
    ></script>

    <zulip-chat
      server="https://chat.example.com"
      auth-token="<short-lived JWT from your backend>"
      channel="general"
      topic="welcome"
      theme="light"
    ></zulip-chat>
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import {ZulipChat} from "zulip-embed-react";

    export function App() {
      return (
        <ZulipChat
          server="https://chat.example.com"
          authToken={authToken}
          channel="general"
          topic="welcome"
          theme="light"
        />
      );
    }
    ```
  </TabItem>
</Tabs>
